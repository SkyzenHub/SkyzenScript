    local function keyImage(url, fname)
        if url == "" or not writefile then return nil end
        local getAsset = getcustomasset or getsynasset
        if not getAsset then return nil end

        local function isImg(d)
            if type(d) ~= "string" or #d < 100 then return false end
            local h = d:sub(1, 4)
            return h == "\137PNG" or h:sub(1, 2) == "\255\216" or h == "RIFF"
        end

        local data
        local req = (syn and syn.request) or request or http_request
        if req then
            local ok, res = pcall(function() return req({Url = url, Method = "GET"}) end)
            if ok and res and isImg(res.Body) then data = res.Body end
        end
        if not data then
            local ok, r = pcall(function() return game:HttpGet(url) end)
            if ok and isImg(r) then data = r end
        end
        if not data then return nil end
        if not pcall(writefile, fname, data) then return nil end
        local ok, id = pcall(getAsset, fname)
        if ok then return id end
        return nil
    end

local hooked = {}
local function hookPrompt(obj)
    if obj:IsA("ProximityPrompt") and obj.Name == "CollectHerbPrompt" then
        obj.HoldDuration = 0
        obj.RequiresLineOfSight = false
    end
end

local function hookSpots(spots)
    if hooked[spots] then return end
    hooked[spots] = true
    for _, d in ipairs(spots:GetDescendants()) do hookPrompt(d) end
    spots.DescendantAdded:Connect(hookPrompt)
end

local function applyPromptUi(obj)
    if obj:IsA("ProximityPrompt") and obj.Name == "CollectHerbPrompt" then
        obj.UIOffset = Vector2.new(0, -CONFIG.PROMPT_UP)
    end
end

local function refreshPromptUi()
    local spots = workspace:FindFirstChild("HerbSpots")
    if not spots then return end

end

local function purgeMap(spots)
    for _, obj in ipairs(workspace:GetChildren()) do
        if obj ~= spots
            and obj ~= workspace.CurrentCamera
            and obj.Name ~= "Terrain"
            and not Players:GetPlayerFromCharacter(obj)
            and (obj:IsA("BasePart") or obj:IsA("Model") or obj:IsA("Folder")) then
            pcall(function() obj:Destroy() end)
        end
    end
end

local function getPos(obj)
    if obj:IsA("Model") then return obj:GetPivot().Position end
    if obj:IsA("BasePart") then return obj.Position end
end

local function findNearTarget(spots, hrp)
    for _, tier in ipairs(activeTiers()) do
        for _, spot in ipairs(spots:GetChildren()) do
            if spot.Parent and spot:GetAttribute("HerbYear") == tier then
                local prompt = spot:FindFirstChild("CollectHerbPrompt", true)
                local pos = getPos(spot)
                if prompt and prompt:IsA("ProximityPrompt") and prompt.Enabled and pos
                    and (pos - hrp.Position).Magnitude <= prompt.MaxActivationDistance + 2 then
                    return spot, prompt, tier
                end
            end
        end
    end
end

local function farmLoop(id)
    local purged = false
    while CONFIG.ENABLED and CONFIG.FARM and farmId == id do
        local spots = workspace:FindFirstChild("HerbSpots")
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")

        if not (spots and hrp) then
            task.wait(1)
        else
            hookSpots(spots)
            if CONFIG.PURGE and not purged then
                purged = true
                purgeMap(spots)
            end

            local spot, prompt = findNearTarget(spots, hrp)
            if spot then
                pcall(function() fireproximityprompt(prompt) end)
                VIM:SendKeyEvent(true, Enum.KeyCode.E, false, game)
                task.wait(0.02)
                VIM:SendKeyEvent(false, Enum.KeyCode.E, false, game)
                task.wait(0.1)
            else
                task.wait(CONFIG.SCAN_DELAY / 1000)
            end
        end
    end
end

local tweenId = 0

local swordFlightEvent = nil

local function moveTo(hrp, pos, id)
    local cf = CFrame.new(pos + Vector3.new(0, 3, 0))
    local dist = (hrp.Position - cf.Position).Magnitude
    if dist < 5 then
        hrp.CFrame = cf
        return true
    end

    local t = math.max(dist / CONFIG.TWEEN_SPEED, 0.02)
    local tw = TS:Create(hrp, TweenInfo.new(t, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut), {CFrame = cf})
    local done = false
    local conn = tw.Completed:Connect(function() done = true end)
    tw:Play()

    while not done and CONFIG.TWEEN_FARM and tweenId == id do
        task.wait(0.01)
    end
    conn:Disconnect()

    if not done then
        tw:Cancel()
        return false
    end
    hrp.CFrame = cf
    return true
end

local function findTweenTarget(spots)
    for _, tier in ipairs(activeTiers()) do
        for _, spot in ipairs(spots:GetChildren()) do
            if spot.Parent and spot:GetAttribute("HerbYear") == tier
                and not (skipUntil[spot] and skipUntil[spot] > os.clock()) then
                local prompt = spot:FindFirstChild("CollectHerbPrompt", true)
                local pos = getPos(spot)
                if prompt and prompt:IsA("ProximityPrompt") and prompt.Enabled and pos then
                    return spot, prompt, pos, tier
                end
            end
        end
    end
end

local function tweenLoop(id)
    local purged = false
    while CONFIG.ENABLED and CONFIG.TWEEN_FARM and tweenId == id do
        local spots = workspace:FindFirstChild("HerbSpots")
        if not spots then
            task.wait(1)
        else
            hookSpots(spots)
            if CONFIG.PURGE and not purged then
                purged = true
                purgeMap(spots)
            end

            local spot, prompt, pos = findTweenTarget(spots)
            if spot then
                local char = LocalPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    if moveTo(hrp, pos, id) and CONFIG.TWEEN_FARM and tweenId == id then
                        task.wait(0.2)

                        local collected = false
                        for _ = 1, 4 do
                            if not (CONFIG.TWEEN_FARM and tweenId == id) then break end
                            if not (spot.Parent and prompt.Parent and prompt.Enabled) then
                                collected = true
                                break
                            end
                            pcall(function() fireproximityprompt(prompt) end)
                            VIM:SendKeyEvent(true, Enum.KeyCode.E, false, game)
                            task.wait(0.02)
                            VIM:SendKeyEvent(false, Enum.KeyCode.E, false, game)
                            pcall(function()
                                prompt:InputHoldBegin()
                                task.wait(0.05)
                                prompt:InputHoldEnd()
                            end)
                            task.wait(0.25)
                        end

                        if not collected and CONFIG.TWEEN_FARM and tweenId == id
                            and spot.Parent and prompt.Parent and prompt.Enabled then
                            skipUntil[spot] = os.clock() + 5
                        end
                    end
                else
                    task.wait(0.5)
                end
            else
                task.wait(CONFIG.SCAN_DELAY / 1000)
            end
        end
    end
end

local function setTweenFarm(v)
    CONFIG.TWEEN_FARM = v
    tweenId = tweenId + 1

    local sf = getSwordFlight()
    if sf then
        pcall(function() sf:FireServer(v) end)
    end

    if v then
        task.spawn(tweenLoop, tweenId)
    end
end

-- Kill Aura (multi-target)
local function getCombatEvent()
    local folder = ReplicatedStorage:FindFirstChild("CultivationEvents")
    return folder and folder:FindFirstChild("CombatHit")
end

local kaId = 0
local kaLastTargets = {}

local function kaGetRoot(plr)
    local char = plr.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
end

local function kaHasForceField(plr)
    local char = plr.Character
    if not char then return false end
    return char:FindFirstChildOfClass("ForceField") ~= nil
end

local function kaFindTargets()
    local myRoot = kaGetRoot(LocalPlayer)
    if not myRoot then return {} end
    local list = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            if not (CONFIG.KA_WHITELIST_FRIENDS and kaHasForceField(plr)) then
                local root = kaGetRoot(plr)
                if root then
                    local d = (root.Position - myRoot.Position).Magnitude
                    if d <= CONFIG.KA_RANGE then
                        list[#list + 1] = { plr = plr, dist = d }
                    end
                end
            end
        end
    end
    table.sort(list, function(a, b) return a.dist < b.dist end)
    return list
end

local function kaFire(target)
    local ev = getCombatEvent()
    if ev then
        pcall(function() ev:FireServer(target, CONFIG.KA_DAMAGE) end)
    end
end

local function kaLoop(id)
    local accum = 0
    while CONFIG.ENABLED and CONFIG.KA_ENABLED and kaId == id do
        local targets = kaFindTargets()

        if #targets == 0 then
            kaLastTargets = {}
            accum = 0
            task.wait(0.1)
        else
            local effectiveHz = CONFIG.KA_FIRE_RATE * CONFIG.KA_SPEED_MULT
            local interval = 1 / effectiveHz

            if CONFIG.KA_INSTANT then
                local seen = {}
                for _, t in ipairs(targets) do
                    seen[t.plr] = true
                    if not kaLastTargets[t.plr] then
                        kaFire(t.plr)
                    end
                end
                kaLastTargets = seen
            end

            local dt = task.wait(interval) or interval
            accum = accum + (dt > 0 and dt or interval)

            if accum >= interval then
                local pending = math.floor(accum / interval)
                if pending < 1 then pending = 1 end
                if pending > CONFIG.KA_MAX_PER_FRAME then pending = CONFIG.KA_MAX_PER_FRAME end
                accum = accum - (pending * interval)

                for _ = 1, pending do
                    for _, t in ipairs(targets) do
                        kaFire(t.plr)
                    end
                end
            end
        end
    end
end

local function setKillAura(v)
    CONFIG.KA_ENABLED = v
    kaId = kaId + 1
    kaLastTargets = {}
    if v then
        task.spawn(kaLoop, kaId)
    end
end

-- ESP Chest (real-time: hilang otomatis saat chest diambil)
local chestEsp = {}
local chestConns = {}
local chestRunId = 0

local CHEST_TYPES = { "Wooden", "Bronze", "Jade", "Celestial" }
local CHEST_COLORS = {
    wooden = Color3.fromRGB(170, 110, 60),
    bronze = Color3.fromRGB(235, 140, 50),
    jade = Color3.fromRGB(70, 230, 140),
    celestial = Color3.fromRGB(120, 200, 255),
}
local CHEST_ATTR_KEYS = { "ChestType", "Type", "Rarity", "Tier", "ChestName", "DisplayName" }
local CHEST_TAKEN_KEYS = { "Collected", "Opened", "Taken", "Claimed", "Looted" }

local function isChestName(name)
    return string.find(string.lower(name), "chest", 1, true) ~= nil
end

-- Jenis chest: dari attribute, atau dari nama chest / nama folder di atasnya
local function chestType(obj)
    for _, k in ipairs(CHEST_ATTR_KEYS) do
        local v = obj:GetAttribute(k)
        if type(v) == "string" then
            local lv = string.lower(v)
            for _, t in ipairs(CHEST_TYPES) do
                if string.find(lv, string.lower(t), 1, true) then return t end
            end
        end
    end
    local cur = obj
    while cur and cur ~= workspace do
        local ln = string.lower(cur.Name)
        for _, t in ipairs(CHEST_TYPES) do
            if string.find(ln, string.lower(t), 1, true) then return t end
        end
        cur = cur.Parent
    end
    return nil
end

local function chestDisplayName(obj)
    local t = chestType(obj)
    if t then return t .. " Chest", t end
    local n = obj.Name:gsub("_", " ")
    n = n:gsub("(%l)(%u)", "%1 %2")
    return n, nil
end

local function chestColorOf(typeName)
    if typeName then
        return CHEST_COLORS[string.lower(typeName)] or Color3.fromRGB(255, 235, 120)
    end
    return Color3.fromRGB(255, 235, 120)
end

local function chestIsCandidate(obj)
    if not (obj:IsA("Model") or obj:IsA("BasePart")) then return false end
    if not isChestName(obj.Name) then return false end
    if Players:GetPlayerFromCharacter(obj) then return false end

    if obj:IsA("Model") then
        -- nama jamak (Chests, ChestSpawns) = wadah, bukan chest
        local ln = string.lower(obj.Name)
        if string.sub(ln, -1) == "s" and string.sub(ln, -2) ~= "ss" then return false end
        -- model yang masih berisi model chest lain = wadah
        for _, d in ipairs(obj:GetDescendants()) do
            if d:IsA("Model") and isChestName(d.Name) then return false end
        end
    else
        -- part di dalam model chest: cukup modelnya saja
        local p = obj.Parent
        if p and p:IsA("Model") and isChestName(p.Name) then return false end
    end
    return true
end

local function chestUnregister(obj)
    local d = chestEsp[obj]
    if d then
        pcall(function() d.bb:Destroy() end)
        chestEsp[obj] = nil
    end
end

local function chestRegister(obj)
    if chestEsp[obj] then return end
    if not obj:IsDescendantOf(workspace) then return end
    if not chestIsCandidate(obj) then return end

    local adornee = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
    if not adornee then return end

    -- jangan dobel kalau ada ESP lain di titik yang sama (chest yang terdiri dari banyak part)
    for _, d in pairs(chestEsp) do
        if d.adornee.Parent and (d.adornee.Position - adornee.Position).Magnitude < 6 then
            return
        end
    end

    local parts, hadVisible = {}, false
    if obj:IsA("BasePart") then
        parts[1] = obj
    else
        for _, p in ipairs(obj:GetDescendants()) do
            if p:IsA("BasePart") then
                parts[#parts + 1] = p
                if #parts >= 40 then break end
            end
        end
    end
    for _, p in ipairs(parts) do
        if p.Transparency < 0.9 then hadVisible = true break end
    end

    local name, tname = chestDisplayName(obj)
    local color = chestColorOf(tname)

    local bb = Instance.new("BillboardGui")
    bb.Name = "ChestESP"
    bb.Size = UDim2.new(0, 190, 0, 24)
    bb.AlwaysOnTop = true
    bb.StudsOffsetWorldSpace = Vector3.new(0, 3, 0)
    bb.Adornee = adornee
    bb.Parent = adornee

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.GothamBold
    label.TextSize = 13
    label.TextColor3 = color
    label.TextStrokeTransparency = 0
    label.Text = name
    label.Parent = bb

    chestEsp[obj] = {
        obj = obj,
        bb = bb,
        label = label,
        adornee = adornee,
        name = name,
        parts = parts,
        hadVisible = hadVisible,
        prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true),
    }
end

local function chestTaken(d)
    local obj = d.obj
    if not obj.Parent or not d.adornee.Parent then return true end
    if not obj:IsDescendantOf(workspace) then return true end
    for _, k in ipairs(CHEST_TAKEN_KEYS) do
        if obj:GetAttribute(k) == true then return true end
    end
    if d.prompt and (not d.prompt.Parent or not d.prompt.Enabled) then return true end
    if d.hadVisible then
        local anyVisible = false
        for _, p in ipairs(d.parts) do
            if p.Parent and p.Transparency < 0.95 then anyVisible = true break end
        end
        if not anyVisible then return true end
    end
    return false
end

local function chestTick()
    local char = LocalPlayer.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    local gone = {}
    for obj, d in pairs(chestEsp) do
        if chestTaken(d) then
            gone[#gone + 1] = obj
        else
            local text = d.name
            if myRoot then
                text = text .. " | " .. math.floor((d.adornee.Position - myRoot.Position).Magnitude) .. "m"
            end
            d.label.Text = text
        end
    end
    for _, obj in ipairs(gone) do chestUnregister(obj) end
end

local function chestFullScan(id)
    local n = 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        n = n + 1
        if n % 400 == 0 then
            task.wait()
            if chestRunId ~= id then return end
        end
        if (obj:IsA("Model") or obj:IsA("BasePart")) and isChestName(obj.Name) then
            chestRegister(obj)
        end
    end
end

local function clearChestEsp()
    for _, c in ipairs(chestConns) do pcall(function() c:Disconnect() end) end
    chestConns = {}
    for obj in pairs(chestEsp) do chestUnregister(obj) end
    chestEsp = {}
end

local function setEspChest(v)
    CONFIG.ESP_CHEST = v
    chestRunId = chestRunId + 1
    clearChestEsp()
    if not v then return end

    local id = chestRunId

    -- Real-time: chest baru muncul -> langsung didaftarkan, chest hilang -> langsung dihapus
    chestConns[#chestConns + 1] = workspace.DescendantAdded:Connect(function(obj)
        if (obj:IsA("Model") or obj:IsA("BasePart")) and isChestName(obj.Name) then
            task.defer(function()
                if chestRunId == id then chestRegister(obj) end
            end)
        end
    end)
    chestConns[#chestConns + 1] = workspace.DescendantRemoving:Connect(function(obj)
        if chestEsp[obj] then chestUnregister(obj) end
    end)

    -- Server mengabari perubahan chest -> cek ulang seketika
    task.spawn(function()
        local folder = ReplicatedStorage:WaitForChild("CultivationEvents", 10)
        local ev = folder and folder:WaitForChild("ChestSync", 5)
        if ev and chestRunId == id then
            chestConns[#chestConns + 1] = ev.OnClientEvent:Connect(function()
                task.defer(function()
                    if chestRunId == id then chestTick() end
                end)
            end)
        end
    end)

    task.spawn(function()
        chestFullScan(id)
        local lastScan = os.clock()
        while CONFIG.ENABLED and CONFIG.ESP_CHEST and chestRunId == id do
            chestTick()
            -- jaring pengaman: scan ulang tiap 8 detik kalau ada yang terlewat
            if os.clock() - lastScan > 8 then
                lastScan = os.clock()
                chestFullScan(id)
            end
            task.wait(0.1)
        end
    end)
end

                end
                if o:IsA("Model") and npcIsCandidate(o) then add(o) end
            end
        end
    end

-- Damage otomatis dari stat player (sumber dicari sekali, lalu hanya dibaca)
local DAMAGE_KEYS = { "damage", "attack", "atk", "dmg", "power", "strength" }
local DAMAGE_SKIP = { "speed", "cooldown", "rate", "chance", "crit", "range", "multi", "boost" }

local function damagePriority(name)
    local n = string.lower(name)
    for _, bad in ipairs(DAMAGE_SKIP) do
        if string.find(n, bad, 1, true) then return nil end
    end
    for i, key in ipairs(DAMAGE_KEYS) do
        if string.find(n, key, 1, true) then return i end
    end
    return nil
end

local function npcFindDamageSource()
    local best, bestP = nil, 99
    local holders = { LocalPlayer, LocalPlayer.Character }
    for _, h in ipairs(holders) do
        if h then
            for k, val in pairs(h:GetAttributes()) do
                if type(val) == "number" and val > 0 then
                    local p = damagePriority(k)
                    if p and p < bestP then
                        best, bestP = { kind = "attr", holder = h, key = k }, p
                    end
                end
            end
            for _, d in ipairs(h:GetDescendants()) do
                if (d:IsA("NumberValue") or d:IsA("IntValue")) and d.Value > 0 then
                    local p = damagePriority(d.Name)
                    if p and p < bestP then
                        best, bestP = { kind = "value", obj = d, key = d.Name }, p
                    end
                end
            end
        end
    end
    return best
end

local function npcReadDamage(src)
    if not src then return nil end
    local v
    if src.kind == "attr" then
        if not src.holder.Parent then return nil end
        v = src.holder:GetAttribute(src.key)
    else
        if not src.obj.Parent then return nil end
        v = src.obj.Value
    end
    if type(v) == "number" and v > 0 then return v end
    return nil
end

local function npcNotify(text)
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "Kill Aura NPC",
            Text = text,
            Duration = 4,
        })
    end)
end

local function npcLoop(id)
    task.spawn(npcScanLoop, id)

    local src = npcFindDamageSource()
    local dmg = npcReadDamage(src)
    if dmg then
        npcNotify("Damage: " .. tostring(dmg) .. " (stat: " .. tostring(src.key) .. ")")
    else
        dmg = 1
        npcNotify("Stat damage tidak ketemu, pakai damage 1")
    end

    local lastDmg, lastFind, lastHit = 0, os.clock(), 0

    while CONFIG.ENABLED and CONFIG.NPC_KA and npcId == id do
        local now = os.clock()

        if now - lastDmg > 0.5 then
            lastDmg = now
            local d = npcReadDamage(src)
            if d then
                dmg = d
            elseif now - lastFind > 5 then
                lastFind = now
                src = npcFindDamageSource()
                local d2 = npcReadDamage(src)
                if d2 then dmg = d2 end
            end
        end

        local char = LocalPlayer.Character
        local myRoot = char and char:FindFirstChild("HumanoidRootPart")
        pcall(npcRingUpdate, myRoot)

        if myRoot and now - lastHit >= CONFIG.NPC_KA_DELAY / 1000 then
            lastHit = now
            local ev = getCombatEvent()
            if ev then
                local myPos = myRoot.Position
                local range = CONFIG.NPC_KA_RANGE
                local list = {}
                for m, part in pairs(npcCandidates) do
                    local dist = (part.Position - myPos).Magnitude
                    if dist <= range then
                        local hum = m:FindFirstChildOfClass("Humanoid")
                        if hum and hum.Health <= 0 then
                            npcCandidates[m] = nil
                        else
                            list[#list + 1] = { inst = m, dist = dist }
                        end
                    end
                end
                table.sort(list, function(a, b) return a.dist < b.dist end)

                for i = 1, math.min(#list, CONFIG.NPC_KA_MAX) do
                    ev:FireServer(list[i].inst, dmg)
                end
            end
        end

        task.wait()
    end
end

local function setKillAuraNpc(v)
    CONFIG.NPC_KA = v
    npcId = npcId + 1
    if v then
        task.spawn(npcLoop, npcId)
    else
        npcCandidates = {}
        npcRingDestroy()
    end
end

local function isImage(data)
    if type(data) ~= "string" or #data < 100 then return false end
    local h = data:sub(1, 4)
    return h == "\137PNG" or h:sub(1, 2) == "\255\216" or h == "RIFF"
end

local function loadImage(url, fname)
    if url == "" then return nil, "empty URL" end
    if not writefile then return nil, "no writefile" end
    local getAsset = getcustomasset or getsynasset
    if not getAsset then return nil, "no getcustomasset" end

    local data
    local req = (syn and syn.request) or request or http_request
    if req then
        local ok, res = pcall(function() return req({Url = url, Method = "GET"}) end)
        if ok and res and isImage(res.Body) then data = res.Body end
    end
    if not data then
        local ok, r = pcall(function() return game:HttpGet(url) end)
        if ok and isImage(r) then data = r end
    end
    if not data then return nil, "download failed" end

    local ok3, err3 = pcall(writefile, fname, data)
    if not ok3 then return nil, "writefile failed: " .. tostring(err3) end
    local ok4, id = pcall(getAsset, fname)
    if not ok4 or not id then return nil, "getcustomasset failed" end
    return id
end

local function addToggle(page, name, default, callback)
    local state = default
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1, -6, 0, 30)
    b.BackgroundColor3 = Color3.fromRGB(55, 55, 64)
    b.BackgroundTransparency = 0.1
    b.Font = Enum.Font.GothamBold
    b.TextSize = 13
    b.TextColor3 = Color3.fromRGB(220, 220, 225)
    b.LayoutOrder = nextOrder(page)
    b.Parent = page
    Instance.new("UICorner", b).CornerRadius = UDim.new(0, 8)
    local s = Instance.new("UIStroke", b)
    b.Text = ""

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    fill.BorderSizePixel = 0
    fill.Parent = b
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 8)

    local shine = Instance.new("UIGradient", fill)
    shine.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(95, 95, 105)),
        ColorSequenceKeypoint.new(0.3, Color3.fromRGB(0, 0, 0)),
        ColorSequenceKeypoint.new(0.7, Color3.fromRGB(0, 0, 0)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(95, 95, 105)),
    })
    local shimmer = nil

    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1, 0, 1, 0)
    txt.BackgroundTransparency = 1
    txt.Font = Enum.Font.GothamBold
    txt.TextSize = 13
    txt.TextColor3 = Color3.new(1, 1, 1)
    txt.Parent = b

    local function refresh(animate)
        txt.Text = name .. "  :  " .. (state and "ON" or "OFF")
        local goalSize = UDim2.new(state and 1 or 0, 0, 1, 0)
        local goalColor = state and Color3.fromRGB(0, 0, 0) or GREY
        if animate then
            TS:Create(fill, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = goalSize}):Play()
            TS:Create(s, TweenInfo.new(0.35), {Color = goalColor, Thickness = state and 2.5 or 1}):Play()
        else
            fill.Size = goalSize
            s.Color = goalColor
            s.Thickness = state and 2.5 or 1
        end
        if state then
            if not shimmer then
                shine.Offset = Vector2.new(-1, 0)
                shimmer = TS:Create(shine, TweenInfo.new(1.4, Enum.EasingStyle.Linear, Enum.EasingDirection.In, -1), {Offset = Vector2.new(1, 0)})
                shimmer:Play()
            end
        elseif shimmer then
            shimmer:Cancel()
            shimmer = nil
        end
    end
    refresh(false)

    b.MouseButton1Click:Connect(function()
        state = not state
        refresh(true)
        pcall(callback, state)
    end)
end

local function makeDraggable(handle, obj)
    local dragging, start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            start = i.Position
            startPos = obj.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - start
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end
